# rexlauncher - in-window launch menu for rexglue recompilation projects.
#
# Add to a game's CMakeLists.txt after `include(generated/rexglue.cmake)`:
#
#   add_subdirectory(../../rexlauncher rexlauncher)   # path to this folder
#   target_link_libraries(<game> PRIVATE rexlauncher)
#
# and override two ReXApp hooks in the game's app class (see README.md).
#
# Links rex::runtime: ImGui and SDL3 resolve to the runtime's own copies, so
# the menu shares its ImGui context and SDL event loop.

cmake_minimum_required(VERSION 3.25)

if(NOT TARGET rex::runtime)
    message(FATAL_ERROR "rexlauncher: rex::runtime not found; add_subdirectory(rexlauncher) "
                        "after include(generated/rexglue.cmake)")
endif()

# Fonts are compiled in (PPSSPP's Roboto Condensed, SIL OFL 1.1, see assets/).
set(_rexlauncher_fonts_cpp "${CMAKE_CURRENT_BINARY_DIR}/rexlauncher_fonts.cpp")
set(_rexlauncher_font_files
    "${CMAKE_CURRENT_SOURCE_DIR}/assets/Roboto_Condensed-Regular.ttf"
    "${CMAKE_CURRENT_SOURCE_DIR}/assets/Roboto_Condensed-Bold.ttf")
set(_rexlauncher_font_symbols kRobotoCondensedRegular kRobotoCondensedBold)

set(_content "// Generated by rexlauncher/CMakeLists.txt from assets/. Do not edit.\n")
string(APPEND _content "#include <cstddef>\n\nnamespace rexlauncher::assets {\n")
foreach(_file _symbol IN ZIP_LISTS _rexlauncher_font_files _rexlauncher_font_symbols)
    file(READ "${_file}" _hex HEX)
    string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," _bytes "${_hex}")
    string(APPEND _content "alignas(4) extern const unsigned char ${_symbol}[] = {${_bytes}};\n")
    string(APPEND _content "extern const std::size_t ${_symbol}Size = sizeof(${_symbol});\n")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_file}")
endforeach()
string(APPEND _content "}  // namespace rexlauncher::assets\n")
file(CONFIGURE OUTPUT "${_rexlauncher_fonts_cpp}" CONTENT "${_content}" @ONLY)

add_library(rexlauncher STATIC
    src/launch_menu.cpp
    src/in_game.cpp
    src/menu.cpp
    src/menu_draw.cpp
    src/input.cpp
    src/keyboard.cpp
    src/keyboard_page.cpp
    src/options.cpp
    src/ports.cpp
    src/settings.cpp
    src/draw.cpp
    src/fonts.cpp
    "${_rexlauncher_fonts_cpp}")
target_include_directories(rexlauncher PUBLIC include)
target_compile_features(rexlauncher PUBLIC cxx_std_23)
target_link_libraries(rexlauncher PUBLIC rex::runtime)
